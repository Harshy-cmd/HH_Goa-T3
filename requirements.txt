# Runtime dependencies (CLI + web app). Dev-only tools live in

# Face detection (YuNet) + face embedding (SFace) both ship inside the OpenCV
# wheel as cv2.FaceDetectorYN / cv2.FaceRecognizerSF. The *headless* wheel drops
# the GUI/libGL system dependencies this project never uses, which keeps slim
# server images (e.g. Render's 512 MB free tier) small and avoids libGL errors.
# Pinned to 4.x: OpenCV 5 has not been validated against this code.
opencv-python-headless==4.12.0.88
numpy>=1.26,<3

# Image decoding + the <=500 KB search-copy re-encode. Pillow 12+ decodes AVIF
# natively; pillow-heif adds HEIC/HEIF (iPhone photos). Together they let the
# pipeline accept any common image format, not just OpenCV's imdecode set.
Pillow>=12.0
pillow-heif>=1.0

requests>=2.31              # search providers + candidate image download

web3==7.16.0                # Ethereum JSON-RPC + contract calls
python-dotenv>=1.0          # .env loading

# Web app: FastAPI backend with Server-Sent Events, served by uvicorn.
